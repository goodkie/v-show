+  processVideoFrame() {
+    if (!this.videoElement || this.videoElement.readyState < 2) return;

+    this.telemetry.previewFrameCount++;
+    const now = Date.now();
+
+    const ctx = this.offscreenCtx;
+    ctx.drawImage(this.videoElement, 0, 0, 160, 120);
+
+    const imgData = ctx.getImageData(0, 45, 160, 30);
+    const data = imgData.data;
+    const currentProfile = new Float32Array(160);
+
+    for (let x = 0; x < 160; x++) {
+      let sum = 0;
+      for (let y = 0; y < 30; y++) {
+        const idx = (y * 160 + x) * 4;
+        sum += 0.299 * data[idx] + 0.587 * data[idx + 1] + 0.114 * data[idx + 2];
+      }
+      currentProfile[x] = sum / 30.0;
+    }
+
+    if (this.prevLuminanceProfile) {
+      let bestShift = 0;
+      let minDiff = Infinity;
+      const testRange = 18;
+      const margin = 20;
+
+      for (let s = -testRange; s <= testRange; s++) {
+        let diff = 0;
+        let count = 0;
+        for (let x = margin; x < 160 - margin; x++) {
+          const targetX = x + s;
+          if (targetX >= 0 && targetX < 160) {
+            diff += Math.abs(currentProfile[targetX] - this.prevLuminanceProfile[x]);
+            count++;
+          }
+        }
+        const meanDiff = count > 0 ? diff / count : Infinity;
+        if (meanDiff < minDiff) {
+          minDiff = meanDiff;
+          bestShift = s;
+        }
+      }
+
+      const sensorsActive = this.telemetry.deviceOrientationEventCount > 4;
+      if (!sensorsActive) {
+        this.telemetry.visualMotionFallbackActive = true;
+        this.telemetry.guidanceMode = 'VISUAL_ONLY';
+
+        if (Math.abs(bestShift) >= 1 && minDiff < 45) {
+          const visualDeltaDeg = -bestShift * 0.406;
+          this.applyMotionDelta(visualDeltaDeg, now);
+        }
+      } else {
+        this.telemetry.guidanceMode = 'SENSOR_VISUAL';
+      }
+    }
+
+    this.prevLuminanceProfile = currentProfile;
+  }
+
+  handleOrientation(e) {
+    if (this.state !== 'CAPTURING') return;
+    this.telemetry.deviceOrientationEventCount++;
+
+    const alpha = e.alpha !== null ? e.alpha : (e.webkitCompassHeading || 0);
+    const beta = e.beta || 0;
+    const gamma = e.gamma || 0;
+    const now = Date.now();
+
+    const pitchDeviation = Math.abs(Math.abs(beta) - 90);
+    const rollDeviation = Math.abs(gamma);
+    if (pitchDeviation > 28 || rollDeviation > 25) {
+      this.levelStatus = 'TILTED';
+      this.guidanceMessage = 'Keep your phone level.';
+    } else {
+      this.levelStatus = 'LEVEL';
+    }
+
+    if (this.lastAngle === null) {
+      this.lastAngle = alpha;
+      this.lastTimestamp = now;
+      this.notifyProgress();
+      return;
+    }
+
+    let delta = alpha - this.lastAngle;
+    if (delta > 180) delta -= 360;
+    if (delta < -180) delta += 360;
+
+    this.applyMotionDelta(delta, now);
+    this.lastAngle = alpha;
+  }
+
+  handleMotion(e) {
+    if (this.state !== 'CAPTURING') return;
+    this.telemetry.deviceMotionEventCount++;
+  }
+
+  applyMotionDelta(delta, now) {
+    const dt = Math.max((now - (this.lastTimestamp || now)) / 1000, 0.016);
+    this.lastTimestamp = now;
+
+    if (!this.directionLocked && Math.abs(delta) > 1.2) {
+      this.rotationDirection = delta > 0 ? 'CLOCKWISE' : 'COUNTERCLOCKWISE';
+      this.directionLocked = true;
+    }
+
+    const currentSpeed = Math.abs(delta) / dt;
+    this.turnSpeedDps = 0.8 * this.turnSpeedDps + 0.2 * currentSpeed;
+
+    if (this.turnSpeedDps > 36.0) {
+      this.speedStatus = 'TOO_FAST';
+      this.guidanceMessage = 'Turn a little slower.';
+    } else if (this.levelStatus === 'LEVEL') {
+      this.speedStatus = 'NORMAL';
+      if (this.accumulatedRotation > 280) {
+        this.guidanceMessage = 'Almost there.';
+      } else {
+        this.guidanceMessage = 'Great — keep going.';
+      }
+    }
+
+    if (this.directionLocked) {
+      const isConsistent = (this.rotationDirection === 'CLOCKWISE' && delta > 0) ||
+                           (this.rotationDirection === 'COUNTERCLOCKWISE' && delta < 0);
+      if (isConsistent) {
+        this.accumulatedRotation += Math.abs(delta);
+      } else if (Math.abs(delta) > 12.0) {
+        this.guidanceMessage = 'Keep turning in the same direction.';
+      }
+    } else if (Math.abs(delta) > 0.5) {
+      this.accumulatedRotation += Math.abs(delta);
+    }
+
+    let stepThreshold = 30.0;
+    if (this.estimatedSceneTexture === 'LOW') stepThreshold = 22.5;
+    else if (this.estimatedSceneTexture === 'HIGH') stepThreshold = 36.0;
+
+    const angleSinceLast = this.accumulatedRotation - (this.lastSelectedAngle || 0);
+    if (angleSinceLast >= stepThreshold && this.canonicalKeyframes.length < this.MAX_CANONICAL_KEYFRAMES) {
+      const sharpness = Math.max(20, Math.round(100 - (this.turnSpeedDps * 0.6)));
+      this.selectCanonicalKeyframe(this.accumulatedRotation, sharpness);
+    }
+
+    if (this.accumulatedRotation >= 330.0 && !this.closureVerified) {
+      this.closureVerified = true;
+    }
+
+    this.notifyProgress();
+
+    if (this.accumulatedRotation >= 355.0 && this.closureVerified) {
+      this.finalizeCapture();
+    }
+  }
+
+  selectCanonicalKeyframe(angle, sharpness) {
+    const keyframe = {
+      index: this.canonicalKeyframes.length + 1,
+      angle: round(angle, 1),
+      timestamp: Date.now(),
+      sharpnessScore: sharpness
+    };
+    this.canonicalKeyframes.push(keyframe);
+    this.lastSelectedAngle = angle;
+    this.lastSelectedTimestamp = Date.now();
+  }
+
+  checkWatchdog() {
+    if (this.state !== 'CAPTURING' && this.state !== 'PREPARING') return;
+
+    const hasVideo = this.telemetry.previewFrameCount > 0 || this.telemetry.videoTrackActive;
+    const hasSensors = this.telemetry.deviceOrientationEventCount > 0;
+
+    if (!hasVideo && !hasSensors) {
+      this.state = 'NEEDS_ATTENTION';
+      this.telemetry.errorCode = 'STARTUP_TIMEOUT_NO_CAMERA_OR_SENSORS';
+      this.notifyAttention({
+        code: 'STARTUP_TIMEOUT',
+        message: 'Camera and motion sensors were not detected within 3.5s. Tap TRY AGAIN or switch to manual photo upload.',
+        diagnosticId: this.diagnosticSessionId
+      });
+      this.sendTelemetry();
+    } else if (hasVideo && this.accumulatedRotation === 0.0) {
+      this.guidanceMessage = 'Camera active! Slowly start turning your body in a full circle.';
+      this.notifyProgress();
+      this.sendTelemetry();
+    }
+  }
+
+  notifyAttention(data) {
+    this.guidanceMessage = data.message;
+    if (this.onAttention) {
+      this.onAttention(data);
+    }
+  }
+
+  finalizeCapture() {
+    this.state = 'COMPLETE';
+    this.stopCaptureResources();
+
+    this.guidanceMessage = 'Circle complete! Processing 360° panorama...';
+    this.notifyProgress();
+    this.sendTelemetry();
+
+    if (this.onComplete) {
+      this.onComplete({
+        accumulatedRotation: this.accumulatedRotation,
+        direction: this.rotationDirection,
+        canonicalKeyframeCount: this.canonicalKeyframes.length,
+        keyframes: this.canonicalKeyframes,
+        closureVerified: this.closureVerified
+      });
+    }
+  }
+
+  notifyProgress() {
+    this.telemetry.progressUpdateCount++;
+    this.telemetry.accumulatedRotation = round(this.accumulatedRotation, 1);
+    const progressPercent = Math.min(100, Math.round((this.accumulatedRotation / 360.0) * 100));
+
+    if (this.onProgress) {
+      this.onProgress({
+        progressPercent,
+        accumulatedRotation: this.accumulatedRotation,
+        speedDps: this.turnSpeedDps,
+        speedStatus: this.speedStatus,
+        levelStatus: this.levelStatus,
+        guidance: this.guidanceMessage,
+        canonicalCount: this.canonicalKeyframes.length,
+        guidanceMode: this.telemetry.guidanceMode
+      });
+    }
+
+    if (progressPercent > 0 && progressPercent % 25 === 0 && !this.telemetryTimer) {
+      this.sendTelemetry();
+      this.telemetryTimer = setTimeout(() => { this.telemetryTimer = null; }, 1000);
+    }
+  }
+
+  sendTelemetry() {
+    try {
+      const projectId = (window.activeProjectData && window.activeProjectData.id) || 
+                        new URLSearchParams(window.location.search).get('projectId') || 
+                        'prj-free-b0c6f3ea';
+      const payload = {
+        ...this.telemetry,
+        accumulatedRotation: round(this.accumulatedRotation, 1),
+        state: this.state,
+        turnSpeedDps: round(this.turnSpeedDps, 1),
+        canonicalKeyframesCount: this.canonicalKeyframes.length,
+        closureVerified: this.closureVerified
+      };
+
+      fetch('/api/projects/' + projectId + '/guided-capture/telemetry', {
+        method: 'POST',
+        headers: { 'Content-Type': 'application/json' },
+        body: JSON.stringify(payload)
+      }).catch(e => console.warn('[GuidedCapture] Telemetry send error:', e));
+    } catch (e) {}
+  }
+
+  stopCaptureResources() {
+    if (this.watchdogTimer) {
+      clearTimeout(this.watchdogTimer);
+      this.watchdogTimer = null;
+    }
+    if (this.animFrameId) {
+      cancelAnimationFrame(this.animFrameId);
+      this.animFrameId = null;
+    }
+    window.removeEventListener('deviceorientation', this.sensorHandler, true);
+    window.removeEventListener('devicemotion', this.motionHandler, true);
+
+    if (this.mediaStream) {
+      try {
+        this.mediaStream.getTracks().forEach(track => track.stop());
+      } catch (e) {}
+      this.mediaStream = null;
+    }
+    if (this.videoElement) {
+      this.videoElement.style.opacity = '0';
+    }
+  }
+
+  stopCapture() {
+    this.state = 'IDLE';
+    this.stopCaptureResources();
+  }
+}

+  renderWheelSegments() {

+    for (let i = 0; i < 8; i++) {
+      const a1 = (i * 45 - 90) * Math.PI / 180;
+      const a2 = ((i + 1) * 45 - 90) * Math.PI / 180;
+      const x1 = cx + rOuter * Math.cos(a1), y1 = cy + rOuter * Math.sin(a1);
+      const x2 = cx + rOuter * Math.cos(a2), y2 = cy + rOuter * Math.sin(a2);
+      const x3 = cx + rInner * Math.cos(a2), y3 = cy + rInner * Math.sin(a2);
+      const x4 = cx + rInner * Math.cos(a1), y4 = cy + rInner * Math.sin(a1);
+      const d = `M ${x1} ${y1} A ${rOuter} ${rOuter} 0 0 1 ${x2} ${y2} L ${x3} ${y3} A ${rInner} ${rInner} 0 0 0 ${x4} ${y4} Z`;
+      out += `<path d="${d}" fill="rgba(56, 189, 248, 0.35)" stroke="#38bdf8" stroke-width="1.5" />`;
+    }
+    return out;

+// Single-use capture session creation using canonical HTTPS origin
+app.post('/api/projects/:id/capture-session/create', async (req, res) => {
+  try {
+    const projectId = req.params.id;
+    // Revoke previous active tokens for this project
+    for (const [t, sess] of captureSessions.entries()) {
+      if (sess.projectId === projectId && sess.status === 'ACTIVE') {
+        sess.status = 'REVOKED';
+        sess.revokedAt = new Date().toISOString();
+      }
+    }
+
+    const sessionToken = 'tok-cap-' + crypto.randomBytes(16).toString('hex');
+    const now = Date.now();
+    const expiresAt = new Date(now + 2 * 60 * 60 * 1000).toISOString(); // 2 hours
+
+    const sessionData = {
+      token: sessionToken,
+      projectId,
+      createdAt: new Date(now).toISOString(),
+      expiresAt,
+      status: 'ACTIVE',
+      usageCount: 0
+    };
+
+    captureSessions.set(sessionToken, sessionData);
+
+    const canonicalUrl = `${CANONICAL_PUBLIC_ORIGIN}/index.html?projectId=${projectId}&token=${sessionToken}&step=6`;
+
+    console.log('[CaptureSession] Created session for ' + projectId + ': ' + sessionToken.slice(0, 12) + '... (expires: ' + expiresAt + ')');
+
+    res.json({
+      ok: true,
+      token: sessionToken,
+      expiresAt,
+      projectId,
+      canonicalUrl
+    });
+  } catch (err) {
+    console.error('[CaptureSession Error]', err);
+    res.status(500).json({ ok: false, error: err.message });
+  }
+});

+app.get('/api/projects/:id/capture-session/:token/verify', (req, res) => {

+    const session = captureSessions.get(req.params.token);
+    if (!session) {
+      return res.status(404).json({ ok: false, valid: false, reason: 'SESSION_NOT_FOUND' });
+    }
+    if (session.status !== 'ACTIVE') {
+      return res.status(403).json({ ok: false, valid: false, reason: 'SESSION_' + session.status });
+    }
+    if (new Date(session.expiresAt).getTime() < Date.now()) {
+      session.status = 'EXPIRED';
+      return res.status(403).json({ ok: false, valid: false, reason: 'SESSION_EXPIRED' });
+    }
+    session.usageCount = (session.usageCount || 0) + 1;
+    res.json({ ok: true, valid: true, session: { ...session, token: session.token.slice(0, 10) + '...' } });
+  } catch (err) {
+    res.status(500).json({ ok: false, error: err.message });
+  }
+});
